CroppedBitmap 裁剪逻辑

从原图裁剪缺陷区域
// 归一化坐标 → 原始像素坐标
pixelX = (int)(defect.X * bitmap.PixelWidth)
pixelY = (int)(defect.Y * bitmap.PixelHeight)
pixelW = (int)(defect.Width * bitmap.PixelWidth)
pixelH = (int)(defect.Height * bitmap.PixelHeight)

// 扩展裁剪边界 (添加 20% padding)
cropX = max(0, pixelX - pixelW * 0.1)
cropY = max(0, pixelY - pixelH * 0.1)
cropW = min(pixelW * 1.2, bitmap.PixelWidth - cropX)
cropH = min(pixelH * 1.2, bitmap.PixelHeight - cropY)

// WPF 裁剪
var cropped = new CroppedBitmap(bitmap,
  new Int32Rect(cropX, cropY, cropW, cropH));

// 缩放到固定显示尺寸 (300x300, 保持比例)
var display = new TransformedBitmap(cropped,
  new ScaleTransform(300.0/cropW, 300.0/cropH));
